4.1向量与矩阵

例子

点积、长度、矩阵乘向量与矩阵乘法，并对照 A B 与 B A。 

import math


def dot(u, v):
    return sum(x * y for x, y in zip(u, v))


def length(u):
    return math.sqrt(dot(u, u))


def mat_vec(M, x):
    return [dot(row, x) for row in M]


def mat_mul(X, Y):
    cols = list(zip(*Y))
    return [[dot(row, col) for col in cols] for row in X]


def f(x):
    s = '%.3f' % x
    return '0.000' if s == '-0.000' else s


def show_vec(v):
    return '[' + ', '.join(f(x) for x in v) + ']'


def show_mat(M):
    return '[' + ', '.join(show_vec(row) for row in M) + ']'


a = [1.0, 2.0]
b = [3.0, 4.0]
A = [[1.0, 2.0], [3.0, 4.0]]
B = [[2.0, 1.0], [0.0, 3.0]]

print('点积 = ' + f(dot(a, b)))
print('a 的长度 = ' + f(length(a)))
print('A b = ' + show_vec(mat_vec(A, b)))
print('A B = ' + show_mat(mat_mul(A, B)))
print('B A = ' + show_mat(mat_mul(B, A)))

def det2(M):
    return M[0][0] * M[1][1] - M[0][1] * M[1][0]


def det3(M):
    a, b, c = M[0]
    m1 = [[M[1][1], M[1][2]], [M[2][1], M[2][2]]]
    m2 = [[M[1][0], M[1][2]], [M[2][0], M[2][2]]]
    m3 = [[M[1][0], M[1][1]], [M[2][0], M[2][1]]]
    return a * det2(m1) - b * det2(m2) + c * det2(m3)


def num(x):
    if abs(x - round(x)) < 1e-9:

    return '%.3f' % x

4.3秩

例子

高斯—约当消元、矩阵的秩，以及方程组解的情况。 

from fractions import Fraction


def rref(M):
    A = [[Fraction(x) for x in row] for row in M]
    rows, cols = len(A), len(A[0])
    pivots = []
    r = 0
    for c in range(cols):
        p = None
        for i in range(r, rows):
            if A[i][c] != 0:
                p = i
                break
        if p is None:
            continue
        A[r], A[p] = A[p], A[r]
        v = A[r][c]
        A[r] = [x / v for x in A[r]]
        for i in range(rows):
            if i != r and A[i][c] != 0:
                k = A[i][c]
                A[i] = [x - k * y for x, y in zip(A[i], A[r])]
        pivots.append(c)
        r += 1
        if r == rows:
            break
    return A, pivots


def show(M):
    return '\n'.join('[' + ', '.join(str(x) for x in row) + ']'
                     for row in M)


def kind(M):
    R, pivots = rref(M)
    n = len(M[0]) - 1
    for row in R:
        if all(x == 0 for x in row[:n]) and row[n] != 0:
            return '无解'
    if len(pivots) < n:
        return '无穷多解'
    return '唯一解'


M = [[1, 2, 0], [2, 4, 1], [0, 0, 1]]
R, pivots = rref(M)
print('行简化：')
print(show(R))
print('秩 = %d' % len(pivots))
print('增广矩阵的解：' + kind([[1, 1, 3], [2, -1, 0]]))

from fractions import Fraction


def transpose(M):
    return [[M[j][i] for j in range(len(M))] for i in range(len(M[0]))]


def inverse(M):
    n = len(M)
    A = []
    for i in range(n):
        A.append([Fraction(M[i][j]) for j in range(n)] +
                 [Fraction(1 if i == j else 0) for j in range(n)])
    for c in range(n):
        p = None
        for i in range(c, n):
            if A[i][c] != 0:
                p = i
                break
        if p is None:
            return None
        A[c], A[p] = A[p], A[c]
        v = A[c][c]
        A[c] = [x / v for x in A[c]]
        for i in range(n):
            if i != c and A[i][c] != 0:
                k = A[i][c]
                A[i] = [x - k * y for x, y in zip(A[i], A[c])]
    return [row[n:] for row in A]


def f(x):
    s = '%.3f' % float(x)
    return '0.000' if s == '-0.000' else s


def show(M):
    return '[' + ', '.join(

例子

旋转、平移、复合变换与逆变换。 

import math


def trans(deg, tx, ty):
    t = math.radians(deg)
    c, s = math.cos(t), math.sin(t)
    return [[c, -s, tx], [s, c, ty], [0.0, 0.0, 1.0]]


def mul(A, B):
    return [[sum(A[i][k] * B[k][j] for k in range(3)) for j in range(3)]
            for i in range(3)]


def apply(T, p):
    return [T[0][0] * p[0] + T[0][1] * p[1] + T[0][2],
            T[1][0] * p[0] + T[1][1] * p[1] + T[1][2]]


def inverse(T):
    R = [[T[0][0], T[0][1]], [T[1][0], T[1][1]]]
    t = [T[0][2], T[1][2]]
    Rt = [[R[0][0], R[1][0]], [R[0][1], R[1][1]]]
    ti = [-(Rt[0][0] * t[0] + Rt[0][1] * t[1]),
          -(Rt[1][0] * t[0] + Rt[1][1] * t[1])]
    return [[Rt[0][0], Rt[0][1], ti[0]],
            [Rt[1][0], Rt[1][1], ti[1]],
            [0.0, 0.0, 1.0]]


def f(x):
    s = '%.3f' % x
    return '0.000' if s == '-0.000' else s


def show_vec(v):
    return '[' + ', '.join(f(x) for x in v) + ']'


p = [1.0, 0.0]
R90 = trans(90, 0, 0)
Tt = trans(0, 2, 1)
T1 = mul(Tt, R90)
T2 = trans(180, 1, 0)

print('旋转 90° = ' + show_vec(apply(R90, p)))

import math


def eig2(M):
    a, b = M[0]
    c, d = M[1]
    t = a + d
    det = a * d - b * c
    s = math.sqrt(max(t * t - 4 * det, 0.0))
    return [(t + s) / 2, (t - s) / 2]


def eigvec2(M, lam):
    a, b = M[0]
    c, d = M[1]
    if abs(b) > 1e-12:
        v = [b, lam - a]
    elif abs(c) > 1e-12:
        v = [lam - d, c]
    elif abs(a - lam) < 1e-12:
        v = [1.0, 0.0]
    else:
        v = [0.0, 1.0]
    n = math.sqrt(v[0] * v[0] + v[1] * v[1])
    v = [v[0] / n, v[1] / n]
    for x in v:
        if abs(x) > 1e-12:
            if x < 0:
                v = [-v[0], -v[1]]
            break
    return v


def power(A, v0, steps=100):
    v = list(v0)
    lam = 0.0
    for _ in range(steps):
        w = [sum(A[i][j] * v[j] for j in range(len(v)))
             for i in range(len(v))]
        lam = sum(v[i] * w[i] for i in range(len(v))) / \
            sum(x * x for x in v)
        n = math.sqrt(sum(x * x for x in w))
        v = [x / n for x in w]
    return lam


A = [[2.0, 1.0], [1.0, 2.0]]
lams = eig2(A)
v1 = eigvec2(A, lams[0])
v2 = eigvec2(A, lams[1])
Q = [[v1[0], v2[0]], [v1[1], v2[1]]]
QlQ = [[sum(Q[i][k] * lams[k] * Q[j][k] for k in range(2))
        for j in range(2)] for i in range(2)]
B = [[4.0, 1.0, 0.0], [1.0, 3.0, 1.0], [0.0, 1.0, 2.0]]

print('特征值 = %.3f, %.3f' % (lams[0], lams[1]))
print('较大特征值的单位特征向量 = [%.3f, %.3f]' % (v1[0], v1[1]))

      (QlQ[0][0], QlQ[0][1], QlQ[1][0], QlQ[1][1]))

def fit(points, through_origin=False):
    n = len(points)
    sx = sum(x for x, _ in points)
    sy = sum(y for _, y in points)
    sxx = sum(x * x for x, _ in points)
    sxy = sum(x * y for x, y in points)
    if through_origin:
        a = sxy / sxx
        b = 0.0
    else:
        det = n * sxx - sx * sx
        a = (n * sxy - sx * sy) / det
        b = (sy - a * sx) / n
    sse = sum((y - (a * x + b)) ** 2 for x, y in points)
    return a, b, sse


points = [(1.0, 2.0), (2.0, 3.0), (3.0, 5.0), (4.0, 7.0)]
a, b, sse = fit(points)

print('a = %.3f, b = %.3f' % (a, b))
print('残差平方和 = %.3f' % sse)
print('x = 5 时的预测值 = %.3f' % (a * 5 + b))

例子

import math


def svd2(M):
    a, b = M[0]
    c, d = M[1]
    p = a * a + c * c
    q = a * b + c * d
    r = b * b + d * d
    t = p + r
    det = p * r - q * q
    disc = math.sqrt(max(t * t - 4 * det, 0.0))
    l1 = (t + disc) / 2
    l2 = (t - disc) / 2
    return math.sqrt(max(l1, 0.0)), math.sqrt(max(l2, 0.0))


def rank1(M):
    a, b = M[0]
    c, d = M[1]
    p = a * a + c * c
    q = a * b + c * d
    r = b * b + d * d
    t = p + r
    det = p * r - q * q
    disc = math.sqrt(max(t * t - 4 * det, 0.0))
    l1 = (t + disc) / 2
    s1 = math.sqrt(max(l1, 0.0))
    if s1 == 0:
        return [[0.0, 0.0], [0.0, 0.0]]
    if abs(q) > 1e-12:
        v = [q, l1 - p]
    elif p >= r:
        v = [1.0, 0.0]
    else:
        v = [0.0, 1.0]
    n = math.sqrt(v[0] * v[0] + v[1] * v[1])
    v = [v[0] / n, v[1] / n]
    w = [a * v[0] + b * v[1], c * v[0] + d * v[1]]
    u = [w[0] / s1, w[1] / s1]
    return [[s1 * u[i] * v[j] for j in range(2)] for i in range(2)]


A = [[4.0, 0.0], [3.0, 5.0]]
s1, s2 = svd2(A)
R = rank1(A)

print('奇异值 = %.3f, %.3f' % (s1, s2))
print('秩 1 近似 = [[%.3f, %.3f], [%.3f, %.3f]]' %
      (R[0][0], R[0][1], R[1][0], R[1][1]))
print('条件数 = %.3f' % (s1 / s2))
